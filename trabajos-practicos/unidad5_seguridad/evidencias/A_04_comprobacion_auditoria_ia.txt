SELECT c.relname AS objeto, COUNT(*) FILTER (WHERE a.grantee = 0) AS permisos_para_public,
       string_agg(DISTINCT pg_get_userbyid(a.grantee), ', ') FILTER (WHERE a.grantee <> 0) AS reciben_permisos
